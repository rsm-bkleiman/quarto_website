Method 4: Shapley Values (Custom Code, Linear Regression, R² value function)

def r2_cache_build(X, y):
    n = X.shape[1]
    cache = {}
    for size in range(n + 1):
        for subset in combinations(range(n), size):
            cache[subset] = (0.0 if len(subset) == 0
                             else sm.OLS(y, sm.add_constant(X[:, list(subset)])).fit().rsquared)
    return cache

def shapley_linear(X, y, names):
    n = len(names)
    cache = r2_cache_build(X, y)
    sv = {}
    for j in range(n):
        others = [i for i in range(n) if i != j]
        s = 0.0
        for size in range(n):
            for subset in combinations(others, size):
                w = math.factorial(size) * math.factorial(n-size-1) / math.factorial(n)
                s += w * (cache[tuple(sorted(subset+(j,)))] - cache[subset])
        sv[names[j]] = s
    return sv, cache

shapley_raw, cache = shapley_linear(X, y, drivers)

Method 5: Johnson’s Relative Weights

def johnsons_rw(X, y):
    Xs = StandardScaler().fit_transform(X)
    ys = StandardScaler().fit_transform(y.reshape(-1,1)).ravel()
    p = Xs.shape[1]
    Rxx = np.corrcoef(Xs.T)
    rxy = np.array([np.corrcoef(Xs[:,j], ys)[0,1] for j in range(p)])
    eigvals, V = np.linalg.eigh(Rxx)
    eigvals = np.maximum(eigvals, 1e-10)
    beta_ols = np.linalg.lstsq(Rxx, rxy, rcond=None)[0]
    beta_z = V.T @ beta_ols
    rw = np.array([
        sum((V[j,k]*np.sqrt(eigvals[k]))**2 * beta_z[k]**2 for k in range(p))
        for j in range(p)
    ])
    return rw, float(rxy @ beta_ols)

Bonus: Shapley Values for Logistic Regression (GLM, McFadden pseudo-R²)

y_bin = (df['satisfaction'] >= 4).astype(int).values

def mcfadden_r2(X_sub, y_bin):
    if X_sub.shape[1] == 0:
        return 0.0
    clf = LogisticRegression(max_iter=500, solver='lbfgs', C=1e6)
    clf.fit(X_sub, y_bin)
    p = np.clip(clf.predict_proba(X_sub)[:,1], 1e-9, 1-1e-9)
    ll_m = np.sum(y_bin*np.log(p) + (1-y_bin)*np.log(1-p))
    p0 = y_bin.mean()
    ll_0 = len(y_bin)*(p0*np.log(p0) + (1-p0)*np.log(1-p0))
    return 1.0 - ll_m / ll_0

def shapley_glm(X, y_bin, names, val_fn):
    n = len(names)
    vc = {}
    for size in range(n+1):
        for subset in combinations(range(n), size):
            vc[subset] = val_fn(X[:, list(subset)], y_bin) if subset else 0.0
    sv = {}
    for j in range(n):
        others = [i for i in range(n) if i != j]
        s = 0.0
        for size in range(n):
            for subset in combinations(others, size):
                w = math.factorial(size)*math.factorial(n-size-1)/math.factorial(n)
                s += w*(vc[tuple(sorted(subset+(j,)))]-vc[subset])
        sv[names[j]] = s
    return sv, vc

shapley_glm_raw, glm_cache = shapley_glm(X, y_bin, drivers, mcfadden_r2)
